function(cortex_warnings TARGET)
	if(MSVC)
		target_compile_options(${TARGET} PRIVATE /W4)
	else()
		# designated initialisers (Platform::Command) leave fields at their defaults on purpose
		target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
	endif()
endfunction()

#### The viewer: html, css and the js files (joined in name order) are embedded in a generated header as raw string literals.
### Editing any of them re-runs this step (CMAKE_CONFIGURE_DEPENDS), so they behave like ordinary sources
function(cortex_embed_viewer HTML CSS JS_FILES HEADER)
	set(TEXT "#pragma once\n// Generated by CMakeLists.txt from the viewer's sources: edit those, not this\n#include <string_view>\nnamespace Viewer\n{\n")
	set(JS "")
	foreach(FILE ${JS_FILES})
		file(READ "${FILE}" PART)
		string(APPEND JS "${PART}\n")
	endforeach()
	file(READ "${HTML}" HTML_TEXT)
	file(READ "${CSS}" CSS_TEXT)
	set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${HTML} ${CSS} ${JS_FILES})
	string(APPEND TEXT "\tinline constexpr std::string_view Html = R\"CORTEXVIEWER(${HTML_TEXT})CORTEXVIEWER\";\n")
	string(APPEND TEXT "\tinline constexpr std::string_view Css = R\"CORTEXVIEWER(${CSS_TEXT})CORTEXVIEWER\";\n")
	string(APPEND TEXT "\tinline constexpr std::string_view Js = R\"CORTEXVIEWER(${JS})CORTEXVIEWER\";\n}\n")
	file(WRITE "${HEADER}.tmp" "${TEXT}")
	configure_file("${HEADER}.tmp" "${HEADER}" COPYONLY) # only rewritten if changed, so nothing recompiles needlessly
endfunction()

#### nlohmann/json: reading and writing the language server's messages (cortex lsp)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
FetchContent_MakeAvailable(json)

#### CorTeX: a library of everything but main, so the tests can use it, and the cortex binary
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS "src/*.cpp")
list(REMOVE_ITEM CORE_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
add_library(cortex_core STATIC ${CORE_SOURCES})
target_include_directories(cortex_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(cortex_core PUBLIC JSL nlohmann_json::nlohmann_json)
cortex_warnings(cortex_core)
file(GLOB VIEWER_JS CONFIGURE_DEPENDS "src/Viewer/js/*.js")
list(SORT VIEWER_JS)
cortex_embed_viewer("${CMAKE_CURRENT_SOURCE_DIR}/src/Viewer/viewer.html" "${CMAKE_CURRENT_SOURCE_DIR}/src/Viewer/viewer.css" "${VIEWER_JS}"
	"${CMAKE_CURRENT_BINARY_DIR}/generated/ViewerSource.h")

add_executable(cortex src/main.cpp)
target_link_libraries(cortex PRIVATE cortex_core)
cortex_warnings(cortex)

#### Tests (make test): Catch2 unit tests of the pure stages, and end-to-end tests which run a cortex binary on tests/fixture
option(CORTEX_TESTS "Build the test suite" OFF)
set(CORTEX_TEST_BINARY "${CMAKE_CURRENT_SOURCE_DIR}/cortex" CACHE FILEPATH "The binary the end-to-end tests run")
if(CORTEX_TESTS)
	FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.12.0)
	FetchContent_MakeAvailable(Catch2)
	file(GLOB_RECURSE TEST_SOURCES CONFIGURE_DEPENDS "tests/*.cpp")
	add_executable(cortex_tests ${TEST_SOURCES})
	target_link_libraries(cortex_tests PRIVATE Catch2::Catch2WithMain cortex_core)
	target_include_directories(cortex_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tests")
	target_compile_definitions(cortex_tests PRIVATE
		CORTEX_BINARY="${CORTEX_TEST_BINARY}"
		CORTEX_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixture"
		CORTEX_TEST_SCRATCH="${CMAKE_CURRENT_BINARY_DIR}/test-scratch")
	set_target_properties(cortex_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
	add_dependencies(cortex_tests cortex)
endif()
